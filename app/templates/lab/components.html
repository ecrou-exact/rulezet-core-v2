Graph Viewer

Network graph powered by Pivotick. JSON config maps node/edge types to shapes and colors. Two modes: simple (viewer) and dev (full UI + filters). Use the button to edit the config/data live.

graph-viewer.js
Usage 
import GraphViewer from '/static/js/components/graph-viewer.js'

<!-- Simple mode: pan/zoom only -->
<graph-viewer :config="cfg" :data="dat" mode="simple" height="400px"></graph-viewer>

<!-- Dev mode: full UI + node type filters + search -->
<graph-viewer :config="cfg" :data="dat" mode="dev" height="500px"></graph-viewer>

<!-- Editable: live JSON editor panel for config + data -->
<graph-viewer :config="cfg" :data="dat" :editable="true" height="600px"></graph-viewer>

/* Config JSON schema */
{
  "graph": { "directed": true },
  "nodes": {
    "id_field": "id",
    "label_field": "label",
    "type_field": "type",
    "default": { "color": "#888", "shape": "circle", "size": 16 },
    "types": {
      "person": { "color": "#4e79a7", "shape": "circle", "size": 22, "label_field": "name" },
      "company": { "color": "#f28e2b", "shape": "hexagon", "size": 28 }
    }
  },
  "edges": {
    "from_field": "from",
    "to_field": "to",
    "label_field": "label",
    "type_field": "type",
    "default": { "color": "#aaa" },
    "types": {
      "works_at":  { "color": "#4e79a7", "dashed": false },
      "founded":   { "color": "#f28e2b", "dashed": true  }
    }
  }
}

/* Data JSON schema */
{
  "nodes": [
    { "id": "1", "type": "person",  "name": "Alice" },
    { "id": "2", "type": "company", "label": "Acme Corp" }
  ],
  "edges": [
    { "from": "1", "to": "2", "type": "works_at", "label": "Employee" }
  ]
}

Chart Viewer

Universal ECharts component — 12 chart types from one JSON format. Switch views with the tab bar, click a data point to see its detail, export as PNG, or go fullscreen.

chart-viewer.js
Usage 
import ChartViewer from '/static/js/components/chart-viewer.js'

<!-- All 12 chart types, click-to-detail enabled -->
<chart-viewer
    :data="my_data"
    :views="['line','area','bar','bar-h','pie','donut','scatter',
             'radar','funnel','gauge','treemap','heatmap','table']"
    :detail="true"
    height="440px">
</chart-viewer>

<!-- Single view — no tab bar -->
<chart-viewer :data="my_data" views="pie" height="320px"></chart-viewer>

/* Universal JSON format (all chart types read the same structure) */
{
    "title":      "My Chart",
    "subtitle":   "Optional subtitle",
    "categories": ["Jan", "Feb", "Mar"],   // X-axis labels (also pie slice names)
    "series": [
        { "name": "Revenue", "values": [120, 200, 150] },
        { "name": "Cost",    "values": [80,  140, 90]  }
    ],
    "meta": {
        "unit":    "€",        // appended to values in tooltips + detail panel
        "x_label": "Month",
        "y_label": "Amount"
    }
}

/* Chart type notes */
// pie / donut / treemap / funnel → first series only; categories = slice labels
// gauge   → first value of first series (0-100 ideal)
// scatter → values can be [[x,y], ...] pairs or flat array (auto-index X)
// heatmap → series[i].values = row of values; categories = X-axis labels
// table   → always available when :detail="true"

Stat Card

KPI metric card with icon, trend indicator, and color variants. 7 colors: brand blue green red orange purple gray. Trend is positive/negative/zero; use trend-inverse for error-type metrics.

stat-card.js
Usage 
import StatCard from '/static/js/components/stat-card.js'
// also link css/components/stat-card.css

<!-- Basic -->
<stat-card label="Total Users" :value="1240" icon="fa-users" color="blue"></stat-card>

<!-- With trend (positive = green, negative = red) -->
<stat-card label="Revenue" :value="12400" unit="€" icon="fa-euro-sign" color="green" :trend="8.3" trend-label="+8% vs last month"></stat-card>

<!-- Inverted trend: down = good (e.g. error count) -->
<stat-card label="Error Rate" :value="2.4" unit="%" icon="fa-circle-xmark" color="red" :trend="-0.8" trend-inverse></stat-card>

<!-- Loading skeleton -->
<stat-card label="API Calls" icon="fa-code" color="gray" :loading="true"></stat-card>

Props:
  value        String|Number|null  Main metric
  label        String              Card title
  icon         String              FA class (without 'fas ')
  color        String              brand | blue | green | red | orange | purple | gray
  unit         String              Appended to value: '%', 'ms', ' GB', etc.
  trend        Number|null         % change — positive/negative/zero
  trend-inverse Boolean            Flip trend color logic (bad if up)
  trend-label  String              Override auto-generated trend text
  sub          String              Small line below the label
  loading      Boolean             Show skeleton state

Timeline

Vertical event timeline with day grouping, level dots, actor avatars, and collapsible descriptions. Emits a select event on item click. Designed for log feeds, activity streams, audit trails.

timeline.js
Usage 
import Timeline from '/static/js/components/timeline.js'
// also link css/components/timeline.css

<timeline :items="logs" :loading="fetching" @select="openDetail"></timeline>

Item shape:
{
  uuid:        String   // unique key
  title:       String   // main text
  level:       String   // 'info' | 'success' | 'warning' | 'error'
  category:    String   // 'user' | 'system' | 'security' | 'admin' | 'api' | 'database' | 'connectors'
  action:      String   // e.g. 'create', 'delete', 'login' (shown as mono tag)
  created_at:  String   // ISO datetime
  actor_name:  String   // null → shows "System"
  description: String   // optional long text (collapsible after max-desc chars)
}

Props:
  items        Array    Required
  loading      Boolean  Show skeleton rows (default: false)
  group-by-day Boolean  Group under Today / Yesterday / date headers (default: true)
  max-desc     Number   Chars before "more" button (default: 180)

Events:
  select(item) — emitted when a row is clicked

ANSI Terminal

Terminal output panel with ANSI escape code rendering. Two modes: view (read-only log display) and edit (interactive input line, emits @submit).

ansi-terminal.js
Usage 
import AnsiTerminal from '/static/js/components/ansi-terminal.js'

<!-- view mode (default) — read-only log panel -->
<ansi-terminal
    :entries="job.logs"
    :live="job.status === 'running'"
    title="Execution logs"
    @clear="job.logs = []">
</ansi-terminal>

<!-- edit mode — adds interactive input line -->
<ansi-terminal
    :entries="terminal_log"
    mode="edit"
    placeholder="Enter command…"
    @submit="on_submit"
    @clear="terminal_log = []">
</ansi-terminal>

// @submit receives the typed text; push feedback into entries
function on_submit(text) {
    terminal_log.push({ ts: new Date().toISOString(), msg: '> ' + text, level: 'info' })
    // run command, push result lines…
}

Props:
  entries      [{ts?, msg, level?}]  Structured log objects
  lines        String[]              Raw ANSI text lines
  mode         String                'view' (default) | 'edit'
  placeholder  String                Input placeholder (edit mode)
  live         Boolean               Auto-scroll on new lines (default: true)
  title        String                Header label
  max-lines    Number                Truncate at N lines (default: 5000)

Events: clear, submit(text)

File Tree

Collapsible file/directory tree with built-in search, collapse/expand all. Two modes: read (navigate + select) and read-plus (inline file preview + download).

file-tree.js
Usage 
import FileTree from '/static/js/components/file-tree.js'

// Read mode — navigate, built-in search + collapse/expand
<file-tree :nodes="tree_nodes" @select="onNodeClick"></file-tree>

// Read-plus mode — same + inline file preview and download button
<file-tree
    :nodes="tree_nodes"
    mode="read-plus"
    :fetch-content="loadContent">
</file-tree>

// fetch-content: async (node) => string | { content, error }
async function loadContent(node) {
    const res = await fetch('/api/.../file?path=' + node.path)
    const d   = await res.json()
    return { content: d.content }
}

Node shape: { name, type: 'file'|'dir', path, ext, children: [] }

Props:
  nodes         Array     Root nodes
  loading       Boolean   Show skeleton
  mode          String    'read' (default) | 'read-plus'
  fetch-content Function  async (node) => string | { content, error }

Events: select(node)
